    ```typescript
    import { create } from 'zustand';

    type VitalitySource = 'crossfader' | 'audio' | 'ml' | 'static' | 'default';

    interface VitalityStore {
      target: number;       // 0.0 - 1.0
      source: VitalitySource;
      setVitality: (value: number, source?: VitalitySource) => void;
    }

    export const useVitalityStore = create<VitalityStore>((set) => ({
      target: 0,
      source: 'default',
      setVitality: (value, source = 'default') =>
        set({ target: Math.max(0, Math.min(1, value)), source }),
    }));
    ```

    **File 2: `src/lib/color-engine.ts`**

    Create a pure function module with zero side effects. Must export:

    1. `lerpHSL(fromH, fromS, fromL, toH, toS, toL, t, hueDirection)` -- returns `hsl(H, S%, L%)` string.
       - `hueDirection`: `'cw' | 'ccw' | 'shortest'`
       - CW: `delta = ((toH - fromH) + 360) % 360; h = (fromH + delta * t) % 360`
       - CCW: `delta = ((fromH - toH) + 360) % 360; h = (fromH - delta * t + 360) % 360`
       - Shortest: pick CW or CCW based on smaller arc

    2. `effectiveVitality(raw, threshold)` -- returns `Math.min(Math.max((raw - threshold) / (1 - threshold), 0), 1)`.

    3. `computeReefColors(vitality: number): ReefColors` -- the main function.

    `ReefColors` interface with 8 string properties: `primary`, `accent`, `secondary`, `highlight`, `bg`, `surface`, `glow`, `text`.

    **Degraded endpoints (vitality=0.0):**
    | Token | H | S | L |
    |-------|---|---|---|
    | primary | 30 | 59 | 53 |
    | accent | 359 | 34 | 63 |
    | secondary | 36 | 42 | 85 |
    | highlight | 43 | 89 | 38 |
    | bg | 30 | 13 | 9 |
    | surface | 24 | 7 | 14 |
    | text | 36 | 16 | 88 |

    **Healthy endpoints (vitality=1.0):**
    | Token | H | S | L |
    |-------|---|---|---|
    | primary | 175 | 80 | 50 |
    | accent | 320 | 75 | 55 |
    | secondary | 220 | 70 | 45 |
    | highlight | 45 | 85 | 60 |
    | bg | 210 | 38 | 8 |
    | surface | 210 | 30 | 12 |
    | text | 180 | 20 | 92 |

    **Stagger thresholds:**
    | Token | Threshold |
    |-------|-----------|
    | bg | 0.0 |
    | surface | 0.0 |
    | primary | 0.2 |
    | glow | 0.2 |
    | text | 0.3 |
    | accent | 0.4 |
    | highlight | 0.7 |
    | secondary | 0.7 |

    **Hue interpolation directions (from RESEARCH.md Pitfall 1):**
    | Token | Direction | Reason |
    |-------|-----------|--------|
    | primary (30->175) | CW | green is natural path to teal |
    | accent (359->320) | CCW | stays in warm red/pink |
    | bg (30->210) | CW | at L=9%, hue invisible anyway |
    | surface (24->210) | CW | same as bg |
    | highlight (43->45) | CW | trivially safe (2deg) |
    | text (36->180) | CW | warm to cool through green-teal |

    **Special handling for `secondary` (36->220):** Do NOT use standard hue interpolation. Instead:
    - Keep saturation at degraded level (42%) until effective vitality > 0.6
    - At effective vitality > 0.6, ramp saturation quickly toward 70%
    - Snap hue to 220 when effective vitality > 0.5 (while saturation is still low enough to hide the jump)
    - This prevents ugly green/purple mid-states

    **Special handling for `glow`:** Same as `primary` interpolation but output `hsla()` with alpha interpolating from 0.2 (degraded) to 0.5 (healthy).

    Each token's interpolation uses `effectiveVitality(rawVitality, threshold)` as the `t` parameter for `lerpHSL`.

    **File 1: `src/hooks/useVitality.ts`**

    Create a `'use client'` hook that runs the rAF animation loop and writes CSS variables. Follow the `useSpectrogram.ts` pattern for refs and cleanup.

    ```typescript
    'use client';
    import { useEffect, useRef } from 'react';
    import { useVitalityStore } from '@/stores/vitality-store';
    import { computeReefColors } from '@/lib/color-engine';
    import type { ReefColors } from '@/lib/color-engine';
    ```

    **Refs (not state):**
    - `currentRef = useRef(0)` -- current animated vitality value
    - `rafRef = useRef<number | null>(null)` -- rAF handle for cleanup
    - `lastWriteRef = useRef(0)` -- timestamp of last CSS write (30fps throttle)
    - `runningRef = useRef(false)` -- guard against double-start (StrictMode)

    **rAF tick function:**
    1. Read target: `const target = useVitalityStore.getState().target;` (no subscription, no re-render)
    2. Lerp: `currentRef.current += (target - currentRef.current) * 0.08;` (ease-out exponential approach, ~300ms to 95% convergence at 60fps)
    3. Snap: `if (Math.abs(target - currentRef.current) < 0.001) currentRef.current = target;`
    4. Throttle CSS writes to 30fps: `if (timestamp - lastWriteRef.current > 33) { ... }`
    5. Inside throttle block: call `computeReefColors(currentRef.current)` then write all 8 CSS vars via `writeCSSVariables(colors)`
    6. Schedule next frame: `rafRef.current = requestAnimationFrame(tick);`

    **writeCSSVariables function (inline or separate):**
    ```typescript
    function writeCSSVariables(colors: ReefColors): void {
      const s = document.documentElement.style;
      s.setProperty('--reef-primary', colors.primary);
      s.setProperty('--reef-accent', colors.accent);
      s.setProperty('--reef-secondary', colors.secondary);
      s.setProperty('--reef-highlight', colors.highlight);
      s.setProperty('--reef-bg', colors.bg);
      s.setProperty('--reef-surface', colors.surface);
      s.setProperty('--reef-glow', colors.glow);
      s.setProperty('--reef-text', colors.text);
    }
    ```

    **useEffect with cleanup:**
    - On mount: guard `if (runningRef.current) return;`, set `runningRef.current = true`, start rAF loop
    - Cleanup: `cancelAnimationFrame(rafRef.current)`, set `runningRef.current = false`
    - Empty dependency array `[]` -- runs once on mount

    **Export:** `export function useVitality(): void` (no return value -- all output via CSS vars). Also re-export `useVitalityStore` for convenience: `export { useVitalityStore } from '@/stores/vitality-store';`

    **File 2: `src/app/providers.tsx`**

    Add the `useVitality()` call inside the existing `Providers` component. This component is already `'use client'`, so the hook is safe.

    ```typescript
    'use client';

    import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
    import { useState } from 'react';
    import { useVitality } from '@/hooks/useVitality';

    export function Providers({ children }: { children: React.ReactNode }) {
      const [queryClient] = useState(
        () =>
          new QueryClient({
            defaultOptions: {
              queries: {
                staleTime: 60 * 1000,
                refetchOnWindowFocus: false,
              },
            },
          })
      );

      // Start vitality animation loop -- writes --reef-* CSS variables via rAF
      useVitality();

      return (
        <QueryClientProvider client={queryClient}>
          {children}
        </QueryClientProvider>
      );
    }
    ```
